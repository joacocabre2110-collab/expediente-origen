# Instalar en el servidor como root con: crontab /opt/canal-automatico/deploy/crontab.txt
# Cambiar 0 8 por el minuto y hora local deseados. El servidor usa TZ del sistema.
0 8 * * * cd /opt/canal-automatico/deploy && docker compose exec -T canal python /app/scripts/diario.py >> /var/log/canal-automatico.log 2>&1
# Comprobación diaria del contenedor: Docker ejecuta también su healthcheck cada 5 minutos.
5 8 * * * cd /opt/canal-automatico/deploy && docker compose ps --status running | grep -q canal || logger -t canal-automatico 'contenedor detenido'
